
add_namespace = modernization

#Increase in foreign ideologies
country_event = {
	id = modernization.1
	title = modernization.1.t
	desc = modernization.1.d
	picture = GFX_report_event_pinghai_rebellion
	
	mean_time_to_happen = {
		days = 300
		modifier = {
				factor = 0.5
				has_stability < 40
			}
		modifier = {
			factor = 1.5
			has_stability > 60
		}
	}
	trigger = {#Occurs only after 15% modernization progress
		has_country_flag = begun_modernization
		check_variable = {
			modernprogress > 15
		}
		check_variable = {
			modernprogress < 99.99
		}
	}
	
	option = { #All a part of our efforts
		name = modernization.1.a
		ai_chance = {
			base = 60
			modifier = {
				add = 1000
				check_variable = {
					modernprogress > 94
				}
			}
			modifier = {
				factor = 0
				has_stability < 34
			}
		}
		add_stability = -0.05
		custom_effect_tooltip = modernization_add_5_tt
		add_to_variable = { modernprogress = 5 }
		random_list = {
			33 = {
				add_popularity = {
					ideology = fascism
					popularity = 0.05
				}
			}
			33 = {
				add_popularity = {
					ideology = communism
					popularity = 0.05
				}
			}
			34 = {
				add_popularity = {
					ideology = democratic
					popularity = 0.05
				}
			}
		}
	}
	option = { #root out these rabble rousers
		name = modernization.1.b
		ai_chance = {
			base = 40
			modifier = {
				factor = 0.5
				has_stability > 60
			}
			modifier = {
				factor = 0
				has_stability > 90
			}
		}
		add_stability = 0.025
	}
}
