mission_to_modernize_category = {
	begin_modernization = {
		available = {
			has_stability > 0.5
		}
		visible = {
			NOT = {
				has_country_flag = begun_modernization
			}
		}
		is_good = yes
		selectable_mission = yes
		complete_effect = {
			custom_effect_tooltip = begun_modernization_tt
			set_country_flag = begun_modernization
			set_variable = {
				var = mptick
				value = 2
			}
			multiply_variable = { mptick = stability }
			set_variable = {
				var = modernprogress
				value = 10
			}
		}
		ai_will_do = {
			factor = 10
		}
	}
	modernization_progress = {
		visible = {
			has_country_flag = begun_modernization
		}
		activation = {
			has_country_flag = begun_modernization
		}
		available = {
			always = no
		}
		is_good = yes
		selectable_mission = yes
		days_mission_timeout = 59
		days_re_enable = 1
		days_remove = 1
		timeout_effect = {
			add_to_variable = { modernprogress = mptick }
			custom_effect_tooltip = modernization_add_tick_tt
			set_variable = {
				var = mptick
				value = 2
			}
			multiply_variable = { mptick = stability }
		}
	}
	placate_nobility = {
		visible = {
			has_country_flag = begun_modernization
		}
		available = {
			has_political_power > 100
			has_stability < 0.75
		}
		is_good = yes
		selectable_mission = yes
		complete_effect = {
			add_political_power = -100
			add_stability = 0.1
			add_war_support = 0.1
		}
		ai_will_do = {
			factor = 5
			modifier = {
				factor = 10
				has_stability < 50
			}
		}
	}
	seize_foreign_assets = {
		visible = {
			has_country_flag = begun_modernization
		}
		is_good = yes
		selectable_mission = yes
		complete_effect = {
			#
		}
	}
	modernize = {
		visible = {
			has_country_flag = begun_modernization
		}
		available = {
			check_variable = {
				modernprogress > 99.99
			}
		}
		is_good = yes
		selectable_mission = yes
		complete_effect = {
			add_research_slot = 1
			remove_ideas = uncivilized_nation
			remove_ideas = partially_modernized_nation
		}
	}
}
